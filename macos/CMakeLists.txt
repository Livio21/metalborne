cmake_minimum_required(VERSION 3.24)
set(CMAKE_OSX_ARCHITECTURES x86_64 CACHE STRING "Rosetta process architecture")
set(CMAKE_OSX_DEPLOYMENT_TARGET 26.0 CACHE STRING "Minimum macOS version")
project(bbport_macos LANGUAGES C CXX OBJCXX)
if(NOT APPLE OR NOT CMAKE_OSX_ARCHITECTURES STREQUAL "x86_64")
    message(FATAL_ERROR "The macOS port executes x86-64 game code through Rosetta; build for x86_64.")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_OBJCXX_STANDARD 23)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(BB_CPU_ONLY "Build the loader/runtime diagnostic without a GPU backend" OFF)
include(FetchContent)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(SDL3 TIMEOUT 120 INACTIVITY_TIMEOUT 20 URL https://codeload.github.com/libsdl-org/SDL/tar.gz/d5af35e3fbb5bb6555ed00e69740d52af2a4e877)
FetchContent_MakeAvailable(SDL3)
set(BB_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
file(GLOB ATRAC9_SOURCES CONFIGURE_DEPENDS "${BB_ROOT}/third_party/LibAtrac9/C/src/*.c")
if(NOT ATRAC9_SOURCES)
    message(FATAL_ERROR "Initialize source dependencies with git submodule update --init --recursive")
endif()
add_library(bbatrac9 STATIC ${ATRAC9_SOURCES})
target_compile_options(bbatrac9 PRIVATE -w)
file(GLOB RUNTIME_SOURCES CONFIGURE_DEPENDS "${BB_ROOT}/src/runtime*.c")
add_library(bbruntime STATIC ${RUNTIME_SOURCES} "${BB_ROOT}/src/platform_macos.c")
target_include_directories(bbruntime PUBLIC "${BB_ROOT}" "${BB_ROOT}/src")
target_compile_options(bbruntime PRIVATE -Wall -Wextra -Werror)
target_link_libraries(bbruntime PUBLIC bbatrac9 SDL3::SDL3-static)
if(BB_CPU_ONLY)
    add_library(bbbackend STATIC cpu_backend.c)
    target_include_directories(bbbackend PRIVATE "${BB_ROOT}")
    set(PROBE_NAME bb-probe-cpu)
    set(SMOKE_SOURCE "")
else()
    include(dependencies.cmake)
    add_subdirectory("${BB_ROOT}/gpu" "${CMAKE_BINARY_DIR}/gpu")
    set(PROBE_NAME bb-probe)
    set(SMOKE_SOURCE "${BB_ROOT}/src/vulkan_smoke.c")
    add_library(bbbackend INTERFACE)
    target_link_libraries(bbbackend INTERFACE bbgpu Vulkan::Vulkan)
endif()
add_executable(${PROBE_NAME} "${BB_ROOT}/src/probe.c" ${SMOKE_SOURCE})
target_link_libraries(${PROBE_NAME} PRIVATE bbruntime bbbackend)
# The renderer resolves runtime hooks from the executable at load time.
target_link_options(${PROBE_NAME} PRIVATE -Wl,-export_dynamic)
target_compile_options(${PROBE_NAME} PRIVATE -Wall -Wextra -Werror)
# Reserve the low TLS segment as in shadPS4's macOS executable. Runtime heap and
# guest mappings occupy separate low ranges; host code stays above them.
function(bb_macos_executable target)
    target_link_options(${target} PRIVATE -Wl,-ld_classic,-no_pie,-no_fixup_chains,-no_huge,-pagezero_size,0x4000,-segaddr,TCB_SPACE,0x4000,-image_base,0x700000000000)
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${BB_ROOT}/out")
endfunction()
bb_macos_executable(${PROBE_NAME})
foreach(check IN ITEMS runtime sema file_mods content)
    add_executable(macos-${check}-check EXCLUDE_FROM_ALL "${BB_ROOT}/tests/test_${check}.c")
    target_link_libraries(macos-${check}-check PRIVATE bbruntime bbbackend)
    target_compile_options(macos-${check}-check PRIVATE -UNDEBUG)
    bb_macos_executable(macos-${check}-check)
endforeach()
target_compile_options(macos-runtime-check PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/test_barrier.h")
add_executable(macos-pad-check EXCLUDE_FROM_ALL "${BB_ROOT}/tests/test_pad.c")
target_link_libraries(macos-pad-check PRIVATE bbruntime)
target_compile_options(macos-pad-check PRIVATE -UNDEBUG)
bb_macos_executable(macos-pad-check)
add_executable(macos-platform-check EXCLUDE_FROM_ALL platform_check.c)
target_link_libraries(macos-platform-check PRIVATE bbruntime bbbackend)
target_compile_options(macos-platform-check PRIVATE -UNDEBUG)
bb_macos_executable(macos-platform-check)
